```typescript
  message: string;
```

### 2. React Query with Optimistic Updates

```typescript
// mutations/useCreateUser.ts
export function useCreateUser() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: api.users.create,
    onSuccess: (newUser) => {
      queryClient.setQueryData<User[]>(['users'], (old) => [
        ...(old ?? []),
        newUser,
      ]);
    },
  });
}
```

### 3. Infinite Queries

```typescript
export function useInfiniteUsers() {
  return useInfiniteQuery({
    queryKey: ['users'],
    queryFn: ({ pageParam = 0 }) =>
      api.users.list({ cursor: pageParam }),
    getNextPageParam: (lastPage) => lastPage.nextCursor,
  });
}
```

## Error Handling

### 1. Global Error Boundary

```typescript
// ErrorBoundary.tsx
export class ErrorBoundary extends Component {
  static getDerivedStateFromError(error: Error) {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, info: ErrorInfo) {
    logError(error, info.componentStack);
  }

  render() {
    if (this.state.hasError) {
      return <ErrorFallback error={this.state.error} />;
    }
    return this.props.children;
  }
}
```

### 2. API Error Handling

```typescript
// utils/api.ts
export class ApiError extends Error {
  constructor(
    message: string,
    public code: string,
    public status: number,
  ) {
    super(message);
  }
}

export async function handleResponse<T>(res: Response): Promise<T> {
  if (!res.ok) {
    const error = await res.json().catch(() => ({}));
    throw new ApiError(
      error.message ?? 'Unknown error',
      error.code ?? 'UNKNOWN',
      res.status,
    );
  }
  return res.json();
}
```

## Form Submission Flow

### 1. React Hook Form + API

```typescript
// forms/UserForm.tsx
export function UserForm({ onSubmit }: Props) {
  const form = useForm<UserInput>({
    resolver: zodResolver(UserSchema),
    defaultValues: { email: '' },
  });

  const mutation = useMutation({
    mutationFn: (data) => api.users.create(data),
    onSuccess: () => {
      toast.success('User created');
      form.reset();
      onSubmit?.();
    },
  });

  return (
    <Form {...form} onSubmit={form.handleSubmit(mutation.mutate)}>
      <input {...form.register('email')} />
      {form.formState.errors.email && (
        <p>{form.formState.errors.email.message}</p>
      )}
      <button disabled={mutation.isPending}>
        {mutation.isPending ? 'Saving...' : 'Save'}
      </button>
    </Form>
  );
}
```

### 2. WebSocket (Live Updates)

```typescript
// hooks/useSocket.ts
export function useSocket<T>(event: string) {
  const [value, set] = useState<T | null>(null);

  useEffect(() => {
    const ws = new WebSocket(`wss://api/events?event=${event}`);

    ws.onmessage = (e) => set(JSON.parse(e.data));

    return () => ws.close();
  }, [event]);

  return value;
}
```

### 3. SSE (Server Sent Events)

```typescript
export function useSSE<T>(url: string) {
  const [data, set] = useState<T | null>(null);

  useEffect(() => {
    const es = new EventSource(url);
    es.onmessage = (e) => set(JSON.parse(e.data));
    return () => es.close();
  }, [url]);
}
```

```
  [x] Shared types package (monorepo) or codegen (OpenAPI)

  [x] Error boundary for crashes
  [x] Loading states for every async operation

  [x] Revalidation after mutations

  [x] Toast for success/error feedback
  [x] Retry logic for network failures
```
